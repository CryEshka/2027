PROVIDER-SW (Альт Сервер 11)
hostnamectl set-hostname provider-sw.ssa.prof; exec bash
hostname -f
useradd master
passwd master

ip link add link ens19 name ens19.25 type vlan id 25
ip link set ens19.25 up
ip addr add 78.23.0.2/26 dev ens19.25
ip route add 0.0.0.0/0 via 78.23.0.1
echo "nameserver 77.88.8.8" > /etc/resolv.conf
ping -c3 ya.ru

apt-get update && apt-get install -y openvswitch
sed -i "s/OVS_REMOVE=yes/OVS_REMOVE=no/g" /etc/net/ifaces/default/options
systemctl enable --now openvswitch.service
reboot

echo "TYPE=eth" > /etc/net/ifaces/ens19/options
cp -r /etc/net/ifaces/ens19 /etc/net/ifaces/ens20
cp -r /etc/net/ifaces/ens19 /etc/net/ifaces/ens21
mkdir /etc/net/ifaces/PROVIDER-SW
cat <<EOF > /etc/net/ifaces/PROVIDER-SW/options
TYPE=ovsbr
HOST='ens19'
EOF
mkdir /etc/net/ifaces/ISP
cat <<EOF > /etc/net/ifaces/ISP/options
TYPE=ovsport
BRIDGE=PROVIDER-SW
VID=25
EOF
echo "78.23.0.2/26" > /etc/net/ifaces/ISP/ipv4address
echo "default via 78.23.0.1" > /etc/net/ifaces/ISP/ipv4route
echo "nameserver 44.44.44.44" > /etc/net/ifaces/ISP/resolv.conf
systemctl restart network

ovs-vsctl add-bond PROVIDER-SW bond0 ens20 ens21
ovs-vsctl set port bond0 lacp=active
ovs-vsctl set Port ens19 trunks=25
ovs-vsctl set Port bond0 trunks=25
ovs-vsctl show
ping -c3 77.88.8.8
